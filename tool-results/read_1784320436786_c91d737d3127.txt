    52→function KpiCard({ title, value, unit, change, changeDir, icon: Icon, color }: {
    53→  title: string; value: string | number; unit?: string;
    54→  change?: number; changeDir?: "up" | "down"; icon: React.ElementType; color: string;
    55→}) {
    56→  return (
    57→    <Card className="border-0 shadow-sm">
    58→      <CardContent className="p-4">
    59→        <div className="flex items-center justify-between mb-2">
    60→          <span className="text-xs text-muted-foreground font-medium">{title}</span>
    61→          <div className={`p-1.5 rounded-lg`} style={{ backgroundColor: color + "15" }}>
    62→            <Icon className="w-3.5 h-3.5" style={{ color }} />
    63→          </div>
    64→        </div>
    65→        <div className="flex items-end gap-1.5">
    66→          <span className="text-2xl font-bold">{value}</span>
    67→          {unit && <span className="text-sm text-muted-foreground mb-0.5">{unit}</span>}
    68→        </div>
    69→        {change !== undefined && changeDir && (
    70→          <div className={`flex items-center gap-0.5 mt-1 text-xs font-medium ${changeDir === "up" ? "text-emerald-600" : "text-red-600"}`}>
    71→            {changeDir === "up" ? <ArrowUpRight className="w-3 h-3" /> : <ArrowDownRight className="w-3 h-3" />}
    72→            {change > 0 ? "+" : ""}{change}%
    73→          </div>
    74→        )}
    75→      </CardContent>
    76→    </Card>
    77→  );
    78→}
