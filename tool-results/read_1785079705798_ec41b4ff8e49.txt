    62→// ─── Helper ─────────────────────────────────────────────────────────────────
    63→function KpiCard({ title, value, unit, change, changeDir, icon: Icon, color }: {
    64→  title: string; value: string | number; unit?: string;
    65→  change?: number; changeDir?: "up" | "down"; icon: React.ElementType; color: string;
    66→}) {
    67→  return (
    68→    <Card className="border-0 shadow-sm">
    69→      <CardContent className="p-4">
    70→        <div className="flex items-center justify-between mb-2">
    71→          <span className="text-xs text-muted-foreground font-medium">{title}</span>
    72→          <div className={`p-1.5 rounded-lg`} style={{ backgroundColor: color + "15" }}>
    73→            <Icon className="w-3.5 h-3.5" style={{ color }} />
    74→          </div>
    75→        </div>
    76→        <div className="flex items-end gap-1.5">
    77→          <span className="text-2xl font-bold">{value}</span>
    78→          {unit && <span className="text-sm text-muted-foreground mb-0.5">{unit}</span>}
    79→        </div>
    80→        {change !== undefined && changeDir && (
    81→          <div className={`flex items-center gap-0.5 mt-1 text-xs font-medium ${changeDir === "up" ? "text-emerald-600" : "text-red-600"}`}>
    82→            {changeDir === "up" ? <ArrowUpRight className="w-3 h-3" /> : <ArrowDownRight className="w-3 h-3" />}
    83→            {change > 0 ? "+" : ""}{change}%
    84→          </div>
    85→        )}
    86→      </CardContent>
    87→    </Card>
    88→  );
    89→}
    90→
    91→function ChartCard({ title, subtitle, children, className }: { title: string; subtitle?: string; children: React.ReactNode; className?: string }) {
    92→  return (
    93→    <Card className={`border-0 shadow-sm ${className || ""}`}>
    94→      <CardHeader className="pb-2 pt-4 px-4">
    95→        <CardTitle className="text-sm font-semibold">{title}</CardTitle>
    96→        {subtitle && <CardDescription className="text-xs">{subtitle}</CardDescription>}
    97→      </CardHeader>
    98→      <CardContent className="px-4 pb-4">{children}</CardContent>
    99→    </Card>
   100→  );
   101→}
