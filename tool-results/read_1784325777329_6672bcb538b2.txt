    53→// ─── Helper ─────────────────────────────────────────────────────────────────
    54→function KpiCard({ title, value, unit, change, changeDir, icon: Icon, color }: {
    55→  title: string; value: string | number; unit?: string;
    56→  change?: number; changeDir?: "up" | "down"; icon: React.ElementType; color: string;
    57→}) {
    58→  return (
    59→    <Card className="border-0 shadow-sm">
    60→      <CardContent className="p-4">
    61→        <div className="flex items-center justify-between mb-2">
    62→          <span className="text-xs text-muted-foreground font-medium">{title}</span>
    63→          <div className={`p-1.5 rounded-lg`} style={{ backgroundColor: color + "15" }}>
    64→            <Icon className="w-3.5 h-3.5" style={{ color }} />
    65→          </div>
    66→        </div>
    67→        <div className="flex items-end gap-1.5">
    68→          <span className="text-2xl font-bold">{value}</span>
    69→          {unit && <span className="text-sm text-muted-foreground mb-0.5">{unit}</span>}
    70→        </div>
    71→        {change !== undefined && changeDir && (
    72→          <div className={`flex items-center gap-0.5 mt-1 text-xs font-medium ${changeDir === "up" ? "text-emerald-600" : "text-red-600"}`}>
    73→            {changeDir === "up" ? <ArrowUpRight className="w-3 h-3" /> : <ArrowDownRight className="w-3 h-3" />}
    74→            {change > 0 ? "+" : ""}{change}%
    75→          </div>
    76→        )}
    77→      </CardContent>
    78→    </Card>
    79→  );
    80→}
    81→
    82→function ChartCard({ title, subtitle, children, className }: { title: string; subtitle?: string; children: React.ReactNode; className?: string }) {

    88→      </CardHeader>
    89→      <CardContent className="px-4 pb-4">{children}</CardContent>
    90→    </Card>
    91→  );
    92→}
