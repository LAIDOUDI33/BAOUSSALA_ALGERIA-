    55→// ─── Helper ─────────────────────────────────────────────────────────────────
    56→function KpiCard({ title, value, unit, change, changeDir, icon: Icon, color }: {
    57→  title: string; value: string | number; unit?: string;
    58→  change?: number; changeDir?: "up" | "down"; icon: React.ElementType; color: string;
    59→}) {
    60→  return (
    61→    <Card className="border-0 shadow-sm">
    62→      <CardContent className="p-4">
    63→        <div className="flex items-center justify-between mb-2">
    64→          <span className="text-xs text-muted-foreground font-medium">{title}</span>
    65→          <div className={`p-1.5 rounded-lg`} style={{ backgroundColor: color + "15" }}>
    66→            <Icon className="w-3.5 h-3.5" style={{ color }} />
    67→          </div>
    68→        </div>
    69→        <div className="flex items-end gap-1.5">
    70→          <span className="text-2xl font-bold">{value}</span>
    71→          {unit && <span className="text-sm text-muted-foreground mb-0.5">{unit}</span>}
    72→        </div>
    73→        {change !== undefined && changeDir && (
    74→          <div className={`flex items-center gap-0.5 mt-1 text-xs font-medium ${changeDir === "up" ? "text-emerald-600" : "text-red-600"}`}>
    75→            {changeDir === "up" ? <ArrowUpRight className="w-3 h-3" /> : <ArrowDownRight className="w-3 h-3" />}
    76→            {change > 0 ? "+" : ""}{change}%
    77→          </div>
    78→        )}
    79→      </CardContent>
    80→    </Card>
    81→  );
    82→}
    83→
    84→function ChartCard({ title, subtitle, children, className }: { title: string; subtitle?: string; children: React.ReactNode; className?: string }) {
    85→  return (
    86→    <Card className={`border-0 shadow-sm ${className || ""}`}>

    88→        <CardTitle className="text-sm font-semibold">{title}</CardTitle>
    89→        {subtitle && <CardDescription className="text-xs">{subtitle}</CardDescription>}
    90→      </CardHeader>
    91→      <CardContent className="px-4 pb-4">{children}</CardContent>
    92→    </Card>
    93→  );
    94→}
