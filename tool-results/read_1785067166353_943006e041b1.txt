    58→// ─── Helper ─────────────────────────────────────────────────────────────────
    59→function KpiCard({ title, value, unit, change, changeDir, icon: Icon, color }: {
    60→  title: string; value: string | number; unit?: string;
    61→  change?: number; changeDir?: "up" | "down"; icon: React.ElementType; color: string;
    62→}) {
    63→  return (
    64→    <Card className="border-0 shadow-sm">
    65→      <CardContent className="p-4">
    66→        <div className="flex items-center justify-between mb-2">
    67→          <span className="text-xs text-muted-foreground font-medium">{title}</span>
    68→          <div className={`p-1.5 rounded-lg`} style={{ backgroundColor: color + "15" }}>
    69→            <Icon className="w-3.5 h-3.5" style={{ color }} />
    70→          </div>
    71→        </div>
    72→        <div className="flex items-end gap-1.5">
    73→          <span className="text-2xl font-bold">{value}</span>
    74→          {unit && <span className="text-sm text-muted-foreground mb-0.5">{unit}</span>}
    75→        </div>
    76→        {change !== undefined && changeDir && (
    77→          <div className={`flex items-center gap-0.5 mt-1 text-xs font-medium ${changeDir === "up" ? "text-emerald-600" : "text-red-600"}`}>
    78→            {changeDir === "up" ? <ArrowUpRight className="w-3 h-3" /> : <ArrowDownRight className="w-3 h-3" />}
    79→            {change > 0 ? "+" : ""}{change}%
    80→          </div>
    81→        )}
    82→      </CardContent>
    83→    </Card>
    84→  );
    85→}
    86→
    87→function ChartCard({ title, subtitle, children, className }: { title: string; subtitle?: string; children: React.ReactNode; className?: string }) {
    88→  return (
    89→    <Card className={`border-0 shadow-sm ${className || ""}`}>
    90→      <CardHeader className="pb-2 pt-4 px-4">
    91→        <CardTitle className="text-sm font-semibold">{title}</CardTitle>
    92→        {subtitle && <CardDescription className="text-xs">{subtitle}</CardDescription>}
    93→      </CardHeader>
    94→      <CardContent className="px-4 pb-4">{children}</CardContent>
    95→    </Card>
    96→  );
    97→}
