    55→    55→function KpiCard({ title, value, unit, change, changeDir, icon: Icon, color }: {
    56→    56→  title: string; value: string | number; unit?: string;
    57→    57→  change?: number; changeDir?: "up" | "down"; icon: React.ElementType; color: string;
    58→    58→}) {
    59→    59→  return (
    60→    60→    <Card className="border-0 shadow-sm">
    61→    61→      <CardContent className="p-4">
    62→    62→        <div className="flex items-center justify-between mb-2">
    63→    63→          <span className="text-xs text-muted-foreground font-medium">{title}</span>
    64→    64→          <div className={`p-1.5 rounded-lg`} style={{ backgroundColor: color + "15" }}>
    65→    65→            <Icon className="w-3.5 h-3.5" style={{ color }} />
    66→    66→          </div>
    67→    67→        </div>
    68→    68→        <div className="flex items-end gap-1.5">
    69→    69→          <span className="text-2xl font-bold">{value}</span>
    70→    70→          {unit && <span className="text-sm text-muted-foreground mb-0.5">{unit}</span>}
    71→    71→        </div>
    72→    72→        {change !== undefined && changeDir && (
    73→    73→          <div className={`flex items-center gap-0.5 mt-1 text-xs font-medium ${changeDir === "up" ? "text-emerald-600" : "text-red-600"}`}>
    74→    74→            {changeDir === "up" ? <ArrowUpRight className="w-3 h-3" /> : <ArrowDownRight className="w-3 h-3" />}
    75→    75→            {change > 0 ? "+" : ""}{change}%
    76→    76→          </div>
    77→    77→        )}
    78→    78→      </CardContent>
    79→    79→    </Card>
    80→    80→  );
    81→    81→}
